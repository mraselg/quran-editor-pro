━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

"Studio Al-Qalam" is a high-precision Quranic DTP (Desktop Publishing) editor
built with:
  - TanStack Start (React meta-framework with file-based routing)
  - Zustand (state management) + zundo (temporal/undo middleware)
  - Tailwind CSS v4
  - Fabric.js (Canvas-based artboard)

The app renders 600+ pages of the full Quran, allowing per-row, per-page,
per-surah, and global overrides of Arabic/Bangla font sizes, positions,
and tajweed symbols.
